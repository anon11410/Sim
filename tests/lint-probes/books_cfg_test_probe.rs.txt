// The fault-injection reachability probe: LEDG-10, as an executed compile
// failure rather than a doc comment.
//
// A .txt file, so cargo never compiles it as part of a normal build.
// tests/lints.sh copies it to tests/_cfg_test_probe.rs, asserts that
// `cargo build --tests` then exits NON-zero AND that the output names E0599,
// and removes it again under the existing trap.
//
// **What this proves.** `Books`'s corruption vocabulary sits behind the crate's
// own test configuration and is visible to this crate's unit tests only. Plan
// 02-05 rests entirely on that boundary: every seeded fault in the negative
// tests is written with a method that no consumer of `sim` can call. An
// integration test under `tests/` reaches this crate exactly as any other
// consumer does, so a probe that calls one from here and fails to compile is the
// executed form of that claim.
//
// **EVERY method in the vocabulary is called, each on its own line.** This
// probe used to name only `corrupt_silent_cash`, which left the other three
// unguarded: moving `corrupt_appended_posting` out of the `#[cfg(test)]` block
// and making it `pub` — to reach it from an integration test, say — would leave
// `corrupt_silent_cash` still unresolvable, so the build would still fail with
// E0599 and check 6 would still print its success line, while a method that
// writes state the public API cannot reach had shipped in the library. One call
// per line means any single one of them resolving turns that line green and
// changes what the compiler says. Guard 7i in tests/lints.sh pins the other
// half: every `corrupt_*` method declared in src/books.rs sits inside a
// `#[cfg(test)]` block, so a sixth one cannot be added outside the gate, and
// guard 7j pins the count so a sixth one cannot be added without a line here.
// `corrupt_silent_stock` was the fifth, added with the goods-conservation
// negative tests, and guard 7j is what refused the commit until this file named
// it.
//
// **The E0599 assertion is load-bearing.** A bare "the build failed" assertion
// would stay green if this probe stopped compiling for an unrelated reason — a
// renamed constructor or a moved config key — and the check would then be
// reporting nothing about the boundary it exists to guard. E0599 is
// specifically "no method named ... found", which is the diagnostic that says
// the method does not exist in a non-test build of the library.
//
// The converse direction is asserted in the crate's own unit tests: the same
// methods ARE callable from `#[cfg(test)] mod corrupt` in src/books.rs, so the
// boundary is proved in both directions rather than one.

use std::path::Path;

use sim::books::{Books, Posting, PostingKind};
use sim::config;
use sim::ids::{Account, FirmId, FirmSlot, GoodId, HouseholdId};

#[test]
fn the_fault_injection_vocabulary_is_reachable_from_an_integration_test() {
    let path = Path::new(env!("CARGO_MANIFEST_DIR")).join("config/baseline.toml");
    let (params, _hash) = config::load(&path).expect("the shipped configuration loads");
    let mut books = Books::new(&params).expect("the shipped endowment sums to the stock");

    let household = Account::Household(HouseholdId(0));
    let firm = Account::Firm(FirmId {
        slot: FirmSlot(0),
        generation: 0,
    });
    let draft = Posting {
        seq: 0,
        kind: PostingKind::Transfer,
        debit: household,
        credit: firm,
        debit_cents: 500,
        credit_cents: 501,
        good: GoodId(0),
        units_out: 0,
        units_in: 0,
        cash_residual_cents: 0,
        goods_residual_units: 0,
    };

    // The five lines the compiler must refuse, each on its own so that any one
    // of them resolving changes the diagnostic:
    //   error[E0599]: no method named `corrupt_silent_cash` found for struct
    //                 `Books` in the current scope
    books.corrupt_silent_cash(household, -1);
    books.corrupt_recorded_cash(firm, household, 100, -1);
    books.corrupt_conserving_deficit(household, firm, 100);
    books.corrupt_silent_stock(firm, GoodId(0), 7);
    books.corrupt_appended_posting(draft);
}
